cmake_minimum_required(VERSION 3.24)
project(KitharaNea VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(native/vendor/JUCE)
add_subdirectory(native/harmony)
juce_add_gui_app(KitharaNea PRODUCT_NAME "KITHÁRA NÉA" COMPANY_NAME "Trucabulles" ICON_BIG "${CMAKE_CURRENT_SOURCE_DIR}/resources/independent/brand/icon.png" NEEDS_WEB_BROWSER FALSE NEEDS_CURL FALSE)
target_sources(KitharaNea PRIVATE native/app/Main.cpp native/app/Core.cpp native/app/Audio.cpp native/app/Analysis.cpp native/app/Sheet.cpp native/app/TimedChart.cpp native/app/UI.cpp native/app/UIViews.cpp native/app/TimelineExtras.cpp native/app/LyricsWindow.cpp native/app/Waveform.cpp native/app/Rhythm.cpp native/app/Localisation.cpp native/app/OptionalModels.cpp native/app/Dialogs.cpp native/app/SheetEditor.cpp native/app/Tests.cpp)
target_include_directories(KitharaNea PRIVATE native/app native/harmony/vendor native/harmony/vendor/ort native/harmony/vendor/soxr/src)
target_compile_definitions(KitharaNea PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_USE_MP3AUDIOFORMAT=1 NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(KitharaNea PRIVATE /utf-8 /bigobj)
target_link_libraries(KitharaNea PRIVATE juce::juce_gui_extra juce::juce_audio_utils juce::juce_dsp juce::juce_cryptography soxr dwmapi psapi normaliz)
