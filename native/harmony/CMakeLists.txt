cmake_minimum_required(VERSION 3.15)
project(kithara_lv_native LANGUAGES C CXX)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/vendor/soxr/cmake/Modules")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WITH_OPENMP OFF CACHE BOOL "" FORCE)
set(WITH_LSR_BINDINGS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory(vendor/soxr)
add_executable(kithara-lv-native main.cpp)
target_include_directories(kithara-lv-native PRIVATE vendor vendor/ort vendor/soxr/src)
target_link_libraries(kithara-lv-native PRIVATE soxr psapi)
if(MSVC)
 target_compile_options(kithara-lv-native PRIVATE /O2 /fp:precise /utf-8)
 target_compile_definitions(kithara-lv-native PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
